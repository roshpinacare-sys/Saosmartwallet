   122→   122→### **שכבה 2: Cryptographic Engine (מנוע קריפטוגרפי)**
   123→   123→
   124→   124→```typescript
   125→   125→// לא רק AES-256-GCM — אלא שכבות הצפנה מרובות
   126→   126→interface CryptoEngine {
   127→   127→  // Master key derivation
   128→   128→  deriveMasterKey: (
   129→   129→    password: string,
   130→   130→    salt: Uint8Array,
   131→   131→    iterations?: number
   132→   132→  ) => Promise<Uint8Array>;
   133→   133→  
   134→   134→  // Multi-layer encryption
   135→   135→  encrypt: (
   136→   136→    plaintext: Uint8Array,
   137→   137→    options: {
   138→   138→      algorithm: 'aes-256-gcm' | 'chacha20-poly1305';
   139→   139→      keyId: string;
   140→   140→      additionalData?: Uint8Array;  // AAD
   141→   141→    }
   142→   142→  ) => Promise<EncryptedPayload>;
   143→   143→  
   144→   144→  // Threshold cryptography (Shamir's Secret Sharing)
   145→   145→  shardSecret: (
   146→   146→    secret: Uint8Array,
   147→   147→    threshold: number,
   148→   148→    totalShares: number
   149→   149→  ) => Promise<Share[]>;
   150→   150→  
   151→   151→  reconstructSecret: (
   152→   152→    shares: Share[]
   153→   153→  ) => Promise<Uint8Array>;
   154→   154→  
   155→   155→  // Zero-knowledge proofs (for authentication without revealing password)
   156→   156→  generateZKProof: (
   157→   157→    password: string,
   158→   158→    challenge: Uint8Array
   159→   159→  ) => Promise<ZKProof>;
   160→   160→  
   161→   161→  verifyZKProof: (
   162→   162→    proof: ZKProof,
   163→   163→    challenge: Uint8Array
   164→   164→  ) => Promise<boolean>;
   165→   165→}
   166→   166→
   167→   167→interface EncryptedPayload {
   168→   168→  version: number;
   169→   169→  algorithm: 'aes-256-gcm' | 'chacha20-poly1305';
   170→   170→  ciphertext: Uint8Array;
   171→   171→  iv: Uint8Array;
   172→   172→  authTag: Uint8Array;
   173→   173→  salt: Uint8Array;
   174→   174→  
   175→   175→  // Multi-layer: encrypted with multiple keys
   176→   176→  layers: {
   177→   177→    keyId: string;
   178→   178→    encryptedKey: Uint8Array;  // Key encrypted with master
   179→   179→  }[];
   180→   180→  
   181→   181→  // Integrity check
   182→   182→  checksum: Uint8Array;  // SHA-256 of plaintext
   183→   183→  
   184→   184→  // Metadata
   185→   185→  createdAt: number;
   186→   186→  expiresAt?: number;
   187→   187→}
   188→   188→```

   190→   190→### **שכבה 3: Policy Engine 2.0 (מנוע מדיניות מתקדם)**
   191→   191→
   192→   192→```typescript
   193→   193→// לא רק rules פשוטים — אלא מנוע מדיניות שלם
   194→   194→interface PolicyEngine {
   195→   195→  // Rule types
   196→   196→  rules: {
   197→   197→    // Static rules (hard-coded, cannot be disabled)
   198→   198→    static: StaticRule[];
   199→   199→    
   200→   200→    // Dynamic rules (user-configurable)
   201→   201→    dynamic: DynamicRule[];
   202→   202→    
   203→   203→    // Contextual rules (based on time, location, behavior)
   204→   204→    contextual: ContextualRule[];
   205→   205→    
   206→   206→    // Composite rules (combinations of other rules)
   207→   207→    composite: CompositeRule[];
   208→   208→  };
   209→   209→  
   210→   210→  // Evaluation pipeline
   211→   211→  evaluate: (
   212→   212→    operation: Operation,
   213→   213→    context: OperationContext
   214→   214→  ) => Promise<PolicyDecision>;
   215→   215→  
   216→   216→  // Anomaly detection
   217→   217→  detectAnomalies: (
   218→   218→    operation: Operation,
   219→   219→    history: OperationHistory
   220→   220→  ) => Promise<AnomalyScore>;
   221→   221→}
   222→   222→
   223→   223→interface OperationContext {
   224→   224→  // Temporal context
   225→   225→  timestamp: number;
   226→   226→  dayOfWeek: number;
   227→   227→  hourOfDay: number;
   228→   228→  
   229→   229→  // Network context
   230→   230→  network: NetworkId;
   231→   231→  blockHeight: number;
   232→   232→  recentTransactions: Transaction[];
   233→   233→  
   234→   234→  // Behavioral context
   235→   235→  userBehavior: {
   236→   236→    averageTransferAmount: bigint;
   237→   237→    averageTransferFrequency: number;
   238→   238→    typicalRecipients: string[];
   239→   239→    lastLoginTime: number;
   240→   240→    loginAttempts: number;
   241→   241→  };
   242→   242→  
   243→   243→  // Risk context
   244→   244→  riskScore: number;  // 0-100
   245→   245→  trustedDevice: boolean;
   246→   246→  knownLocation: boolean;
   247→   247→}
   248→   248→
   249→   249→interface PolicyDecision {
   250→   250→  allowed: boolean;
   251→   251→  confidence: number;  // 0-1
   252→   252→  reasons: string[];
   253→   253→  
   254→   254→  // If not allowed, what's needed
   255→   255→  requirements?: {
   256→   256→    additionalApprovals?: string[];
   257→   257→    multiSigThreshold?: number;
   258→   258→    timeLock?: number;  // seconds to wait
   259→   259→    stepUpAuth?: boolean;  // require biometric/password re-entry
   260→   260→  };
   261→   261→  
   262→   262→  // Risk assessment
   263→   263→  riskLevel: 'low' | 'medium' | 'high' | 'critical';
   264→   264→  anomalies: Anomaly[];
   265→   265→}
   266→   266→
   267→   267→// Example: Contextual rule
   268→   268→interface ContextualRule {
   269→   269→  id: string;
   270→   270→  name: string;
   271→   271→  
   272→   272→  condition: {
   273→   273→    // Time-based
   274→   274→    timeWindow?: { start: string; end: string };  // "09:00-17:00"
   275→   275→    
   276→   276→    // Amount-based
   277→   277→    amountThreshold?: bigint;
   278→   278→    
   279→   279→    // Frequency-based
   280→   280→    maxOperationsPerHour?: number;
   281→   281→    
   282→   282→    // Recipient-based
   283→   283→    newRecipientCooldown?: number;  // hours before new recipient is trusted
   284→   284→    
   285→   285→    // Location-based
   286→   286→    requireKnownLocation?: boolean;
   287→   287→  };
   288→   288→  
   289→   289→  action: 'allow' | 'deny' | 'require_approval' | 'step_up_auth';
   290→   290→}
   291→   291→```

   293→   293→### **שכבה 4: Signing Orchestrator (מנצח חתימות)**
   294→   294→
   295→   295→```typescript
   296→   296→// לא רק חתימה בודדת — אלא תזמור שלם
   297→   297→interface SigningOrchestrator {
   298→   298→  // Single signature
   299→   299→  sign: (
   300→   300→    vaultId: string,
   301→   301→    message: Uint8Array,
   302→   302→    options?: SigningOptions
   303→   303→  ) => Promise<Signature>;
   304→   304→  
   305→   305→  // Multi-signature (for multi-sig vaults)
   306→   306→  multiSign: (
   307→   307→    vaultId: string,
   308→   308→    message: Uint8Array,
   309→   309→    requiredSigners: string[]
   310→   310→  ) => Promise<MultiSigResult>;
   311→   311→  
   312→   312→  // Batch signing (for multiple operations)
   313→   313→  batchSign: (
   314→   314→    vaultId: string,
   315→   315→    operations: Operation[]
   316→   316→  ) => Promise<BatchSignResult>;
   317→   317→  
   318→   318→  // Cross-chain signing (for bridge operations)
   319→   319→  crossChainSign: (
   320→   320→    sourceVaultId: string,
   321→   321→    targetNetwork: NetworkId,
   322→   322→    operation: BridgeOperation
   323→   323→  ) => Promise<CrossChainSignature>;
   324→   324→  
   325→   325→  // Transaction simulation (before actual signing)
   326→   326→  simulate: (
   327→   327→    operation: Operation

   330→   330→
   331→   331→interface SigningOptions {

   335→   335→  
   336→   336→  // Time-lock (transaction can only be broadcast after X time)
   337→   337→  timeLock?: number;
   338→   338→  
   339→   339→  // Fee limits
   340→   340→  maxFee?: bigint;
   341→   341→  
   342→   342→  // Confirmation callback
   343→   343→  onConfirmationRequired?: (details: ConfirmationDetails) => Promise<boolean>;
   344→   344→}
   345→   345→
   346→   346→interface SimulationResult {
   347→   347→  success: boolean;
   348→   348→  
   349→   349→  // What would happen
   350→   350→  expectedOutcome: {
   351→   351→    balanceChanges: { asset: string; amount: bigint }[];
   352→   352→    stateChanges: string[];
   353→   353→    events: string[];
   354→   354→  };
   355→   355→  
   356→   356→  // Risk assessment
   357→   357→  risks: {
   358→   358→    type: 'slippage' | 'front_running' | 'insufficient_balance' | 'policy_violation';
   359→   359→    severity: 'low' | 'medium' | 'high';
   360→   360→    description: string;
   361→   361→  }[];
   362→   362→  
   363→   363→  // Cost estimation
   364→   364→  estimatedFee: bigint;
   365→   365→  estimatedResourceCredits: number;
   366→   366→}
   367→   367→```

   416→   416→    index: number;
   417→   417→    data: Uint8Array;
   418→   418→  }[];
   419→   419→  
   420→   420→  // Metadata (unencrypted)

   451→   451→### **שכבה 6: Audit & Compliance (ביקורת ותאימות)**
   452→   452→
   453→   453→```typescript
   454→   454→// כל פעולה מתועדת באופן בלתי הפיך
   455→   455→interface AuditSystem {
   456→   456→  // Log every operation
   457→   457→  log: (
   458→   458→    event: AuditEvent
   459→   459→  ) => Promise<void>;
   460→   460→  
   461→   461→  // Query logs
   462→   462→  query: (
   463→   463→    filters: AuditQueryFilters
   464→   464→  ) => Promise<AuditEvent[]>;
   465→   465→  
   466→   466→  // Generate compliance report
   467→   467→  generateReport: (
   468→   468→    options: {
   469→   469→      period: { start: number; end: number };
   470→   470→      format: 'json' | 'csv' | 'pdf';
   471→   471→      includeSensitive?: boolean;
   472→   472→    }
   473→   473→  ) => Promise<ComplianceReport>;
   474→   474→  
   475→   475→  // Tamper detection
   476→   476→  verifyIntegrity: () => Promise<boolean>;
   477→   477→}
   478→   478→
   479→   479→interface AuditEvent {
   480→   480→  id: string;
   481→   481→  timestamp: number;
   482→   482→  
   483→   483→  // What happened
   484→   484→  type: 'vault_created' | 'key_added' | 'signing_attempt' | 
   485→   485→        'policy_violation' | 'backup_created' | 'recovery_initiated';
   486→   486→  
   487→   487→  // Who did it
   488→   488→  actor: {
   489→   489→    vaultId?: string;
   490→   490→    ipAddress?: string;
   491→   491→    deviceId?: string;
   492→   492→    location?: string;
   493→   493→  };
   494→   494→  
   495→   495→  // What was affected
   496→   496→  target: {
   497→   497→    vaultId?: string;
   498→   498→    keyId?: string;
   499→   499→    operation?: Operation;
   500→   500→  };
   501→   501→  
   502→   502→  // Result
   503→   503→  result: 'success' | 'failure' | 'blocked';
   504→   504→  reason?: string;
   505→   505→  
   506→   506→  // Cryptographic proof
   507→   507→  previousEventHash: string;
   508→   508→  eventHash: string;  // SHA-256(prev_hash + event_data)
   509→   509→}
   510→   510→
   511→   511→// Blockchain-like integrity
   512→   512→interface AuditChain {
   513→   513→  genesisHash: string;
   514→   514→  events: AuditEvent[];
   515→   515→  
   516→   516→  // Each event's hash includes previous hash (tamper-evident)
