    41→### **שכבה 1: Vault Hierarchy (היררכיית כספות)**

    48→    id: 'master';

    50→    keyType: 'ed25519';  // לא secp256k1 — יותר מהיר ובטוח
    51→    requiresMultiSig: true;
    52→    threshold: 3;  // 3 מתוך 5 שומרים
    53→  };
    54→  
    55→  // Network Vaults — אחד לכל רשת

    68→

   122→### **שכבה 2: Cryptographic Engine (מנוע קריפטוגרפי)**
   123→
   124→```typescript
   125→// לא רק AES-256-GCM — אלא שכבות הצפנה מרובות
   126→interface CryptoEngine {
   127→  // Master key derivation
   128→  deriveMasterKey: (
   129→    password: string,
   130→    salt: Uint8Array,
   131→    iterations?: number
   132→  ) => Promise<Uint8Array>;
   133→  
   134→  // Multi-layer encryption
   135→  encrypt: (
   136→    plaintext: Uint8Array,
   137→    options: {
   138→      algorithm: 'aes-256-gcm' | 'chacha20-poly1305';
   139→      keyId: string;
   140→      additionalData?: Uint8Array;  // AAD
   141→    }
   142→  ) => Promise<EncryptedPayload>;
   143→  
   144→  // Threshold cryptography (Shamir's Secret Sharing)
   145→  shardSecret: (
   146→    secret: Uint8Array,
   147→    threshold: number,
   148→    totalShares: number
   149→  ) => Promise<Share[]>;
   150→  
   151→  reconstructSecret: (
   152→    shares: Share[]
   153→  ) => Promise<Uint8Array>;
   154→  
   155→  // Zero-knowledge proofs (for authentication without revealing password)
   156→  generateZKProof: (
   157→    password: string,
   158→    challenge: Uint8Array
   159→  ) => Promise<ZKProof>;
   160→  
   161→  verifyZKProof: (
   162→    proof: ZKProof,
   163→    challenge: Uint8Array
   164→  ) => Promise<boolean>;
   165→}
   166→
   167→interface EncryptedPayload {
   168→  version: number;
   169→  algorithm: 'aes-256-gcm' | 'chacha20-poly1305';
   170→  ciphertext: Uint8Array;
   171→  iv: Uint8Array;
   172→  authTag: Uint8Array;
   173→  salt: Uint8Array;
   174→  
   175→  // Multi-layer: encrypted with multiple keys
   176→  layers: {
   177→    keyId: string;
   178→    encryptedKey: Uint8Array;  // Key encrypted with master
   179→  }[];
   180→  
   181→  // Integrity check
   182→  checksum: Uint8Array;  // SHA-256 of plaintext
   183→  
   184→  // Metadata
   185→  createdAt: number;
   186→  expiresAt?: number;
   187→}
   188→```

   190→### **שכבה 3: Policy Engine 2.0 (מנוע מדיניות מתקדם)**
   191→
   192→```typescript
   193→// לא רק rules פשוטים — אלא מנוע מדיניות שלם
   194→interface PolicyEngine {
   195→  // Rule types
   196→  rules: {
   197→    // Static rules (hard-coded, cannot be disabled)
   198→    static: StaticRule[];
   199→    
   200→    // Dynamic rules (user-configurable)
   201→    dynamic: DynamicRule[];
   202→    
   203→    // Contextual rules (based on time, location, behavior)
   204→    contextual: ContextualRule[];
   205→    
   206→    // Composite rules (combinations of other rules)
   207→    composite: CompositeRule[];
   208→  };
   209→  
   210→  // Evaluation pipeline
   211→  evaluate: (
   212→    operation: Operation,
   213→    context: OperationContext
   214→  ) => Promise<PolicyDecision>;
   215→  
   216→  // Anomaly detection
   217→  detectAnomalies: (
   218→    operation: Operation,
   219→    history: OperationHistory
   220→  ) => Promise<AnomalyScore>;
   221→}
   222→
   223→interface OperationContext {
   224→  // Temporal context
   225→  timestamp: number;
   226→  dayOfWeek: number;
   227→  hourOfDay: number;
   228→  
   229→  // Network context
   230→  network: NetworkId;
   231→  blockHeight: number;
   232→  recentTransactions: Transaction[];
   233→  
   234→  // Behavioral context
   235→  userBehavior: {
   236→    averageTransferAmount: bigint;
   237→    averageTransferFrequency: number;
   238→    typicalRecipients: string[];
   239→    lastLoginTime: number;
   240→    loginAttempts: number;
   241→  };
   242→  
   243→  // Risk context
   244→  riskScore: number;  // 0-100
   245→  trustedDevice: boolean;
   246→  knownLocation: boolean;
   247→}
   248→
   249→interface PolicyDecision {
   250→  allowed: boolean;
   251→  confidence: number;  // 0-1
   252→  reasons: string[];
   253→  
   254→  // If not allowed, what's needed
   255→  requirements?: {
   256→    additionalApprovals?: string[];
   257→    multiSigThreshold?: number;
   258→    timeLock?: number;  // seconds to wait
   259→    stepUpAuth?: boolean;  // require biometric/password re-entry
   260→  };
   261→  
   262→  // Risk assessment
   263→  riskLevel: 'low' | 'medium' | 'high' | 'critical';
   264→  anomalies: Anomaly[];
   265→}
   266→
   267→// Example: Contextual rule
   268→interface ContextualRule {
   269→  id: string;
   270→  name: string;
   271→  
   272→  condition: {
   273→    // Time-based
   274→    timeWindow?: { start: string; end: string };  // "09:00-17:00"
   275→    
   276→    // Amount-based
   277→    amountThreshold?: bigint;
   278→    
   279→    // Frequency-based
   280→    maxOperationsPerHour?: number;
   281→    
   282→    // Recipient-based
   283→    newRecipientCooldown?: number;  // hours before new recipient is trusted
   284→    
   285→    // Location-based
   286→    requireKnownLocation?: boolean;
   287→  };
   288→  
   289→  action: 'allow' | 'deny' | 'require_approval' | 'step_up_auth';
   290→}
   291→```

   293→### **שכבה 4: Signing Orchestrator (מנצח חתימות)**
   294→
   295→```typescript
   296→// לא רק חתימה בודדת — אלא תזמור שלם
   297→interface SigningOrchestrator {
   298→  // Single signature
   299→  sign: (
   300→    vaultId: string,
   301→    message: Uint8Array,
   302→    options?: SigningOptions
   303→  ) => Promise<Signature>;
   304→  
   305→  // Multi-signature (for multi-sig vaults)
   306→  multiSign: (
   307→    vaultId: string,
   308→    message: Uint8Array,
   309→    requiredSigners: string[]
   310→  ) => Promise<MultiSigResult>;
   311→  
   312→  // Batch signing (for multiple operations)
   313→  batchSign: (
   314→    vaultId: string,
   315→    operations: Operation[]
   316→  ) => Promise<BatchSignResult>;
   317→  
   318→  // Cross-chain signing (for bridge operations)
   319→  crossChainSign: (
   320→    sourceVaultId: string,
   321→    targetNetwork: NetworkId,
   322→    operation: BridgeOperation
   323→  ) => Promise<CrossChainSignature>;
   324→  
   325→  // Transaction simulation (before actual signing)
   326→  simulate: (
   327→    operation: Operation
   328→  ) => Promise<SimulationResult>;
   329→}
   330→
   331→interface SigningOptions {
   332→  // Require additional authentication
   333→  requireBiometric?: boolean;
   334→  requirePassword?: boolean;
   335→  
   336→  // Time-lock (transaction can only be broadcast after X time)
   337→  timeLock?: number;
   338→  
   339→  // Fee limits
   340→  maxFee?: bigint;
   341→  
   342→  // Confirmation callback
   343→  onConfirmationRequired?: (details: ConfirmationDetails) => Promise<boolean>;
   344→}
   345→
   346→interface SimulationResult {
   347→  success: boolean;
   348→  
   349→  // What would happen
   350→  expectedOutcome: {
   351→    balanceChanges: { asset: string; amount: bigint }[];
   352→    stateChanges: string[];
   353→    events: string[];
   354→  };
   355→  
   356→  // Risk assessment
   357→  risks: {
   358→    type: 'slippage' | 'front_running' | 'insufficient_balance' | 'policy_violation';
   359→    severity: 'low' | 'medium' | 'high';
   360→    description: string;
   361→  }[];
   362→  
   363→  // Cost estimation
   364→  estimatedFee: bigint;
   365→  estimatedResourceCredits: number;
   366→}
   367→```

   369→### **שכבה 5: Backup & Recovery (גיבוי ושחזור)**
   370→
   371→```typescript
   372→// לא רק export/import — אלא אסטרטגיית שחזור מלאה
   373→interface BackupStrategy {
   374→  // Encrypted backup
   375→  createBackup: (
   376→    options: {
   377→      includeMasterKey: boolean;
   378→      shardThreshold?: number;  // Shamir's Secret Sharing
   379→      totalShards?: number;
   380→    }
   381→  ) => Promise<BackupPackage>;
   382→  
   383→  // Restore from backup
   384→  restoreBackup: (
   385→    backup: BackupPackage,
   386→    shards?: Share[]  // If sharded
   387→  ) => Promise<void>;
   388→  
   389→  // Social recovery (trusted contacts)
   390→  setupSocialRecovery: (
   391→    contacts: RecoveryContact[],
   392→    threshold: number
   393→  ) => Promise<void>;
   394→  
   395→  initiateSocialRecovery: () => Promise<RecoverySession>;
   396→  
   397→  // Dead-man switch (if no activity for X days)
   398→  setupDeadManSwitch: (
   399→    inactiveDays: number,
   400→    beneficiaries: Beneficiary[]
   401→  ) => Promise<void>;
   402→  
   403→  // Air-gapped backup (QR codes, paper wallets)
   404→  createAirGappedBackup: () => Promise<QRCode[]>;
   405→}
   406→
   407→interface BackupPackage {
   408→  version: number;
   409→  createdAt: number;
   410→  
   411→  // Encrypted with master password
   412→  encryptedData: Uint8Array;
   413→  
   414→  // Sharded master key (if applicable)
   415→  shards?: {
   416→    index: number;
   417→    data: Uint8Array;
   418→  }[];
   419→  
   420→  // Metadata (unencrypted)
   421→  metadata: {
   422→    networks: NetworkId[];
   423→    vaultCount: number;
   424→    keyCount: number;
   425→    checksum: string;
   426→  };
   427→  
   428→  // Integrity verification
   429→  signature: Uint8Array;  // Signed with master key
   430→}
   431→
   432→interface RecoveryContact {
   433→  id: string;
   434→  name: string;
   435→  
   436→  // How to reach them
   437→  contactMethod: 'email' | 'phone' | 'on_chain' | 'in_person';
   438→  contactInfo: string;
   439→  
   440→  // Their public key for encryption
   441→  publicKey: string;
   442→  
   443→  // Trust level
   444→  trustLevel: 'high' | 'medium' | 'low';
   445→  
   446→  // Last verification
   447→  lastVerified: number;
   448→}
   449→```

   451→### **שכבה 6: Audit & Compliance (ביקורת ותאימות)**
   452→
   453→```typescript
   454→// כל פעולה מתועדת באופן בלתי הפיך
   455→interface AuditSystem {
   456→  // Log every operation
   457→  log: (
   458→    event: AuditEvent
   459→  ) => Promise<void>;
   460→  
   461→  // Query logs
   462→  query: (
   463→    filters: AuditQueryFilters
   464→  ) => Promise<AuditEvent[]>;
   465→  
   466→  // Generate compliance report
   467→  generateReport: (
   468→    options: {
   469→      period: { start: number; end: number };
   470→      format: 'json' | 'csv' | 'pdf';
   471→      includeSensitive?: boolean;
   472→    }
   473→  ) => Promise<ComplianceReport>;
   474→  
   475→  // Tamper detection
   476→  verifyIntegrity: () => Promise<boolean>;
   477→}
   478→
   479→interface AuditEvent {
   480→  id: string;
   481→  timestamp: number;
   482→  
   483→  // What happened
   484→  type: 'vault_created' | 'key_added' | 'signing_attempt' | 
   485→        'policy_violation' | 'backup_created' | 'recovery_initiated';
   486→  
   487→  // Who did it
   488→  actor: {
   489→    vaultId?: string;
   490→    ipAddress?: string;
   491→    deviceId?: string;
   492→    location?: string;
   493→  };
   494→  
   495→  // What was affected
   496→  target: {
   497→    vaultId?: string;
   498→    keyId?: string;
   499→    operation?: Operation;
   500→  };
   501→  
   502→  // Result
   503→  result: 'success' | 'failure' | 'blocked';
   504→  reason?: string;
   505→  
   506→  // Cryptographic proof
   507→  previousEventHash: string;
   508→  eventHash: string;  // SHA-256(prev_hash + event_data)
   509→}
   510→
   511→// Blockchain-like integrity
   512→interface AuditChain {
   513→  genesisHash: string;
   514→  events: AuditEvent[];
   515→  
   516→  // Each event's hash includes previous hash (tamper-evident)
   517→  verifyChain: () => boolean;
   518→}
   519→```
